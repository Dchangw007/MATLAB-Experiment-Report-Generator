二、实验代码

基本操作实现函数代码如下（MATLAB）：

clearvars;
close all;

n = 100;
iter = 500;

A = 8*diag(ones(n,1)) + diag(ones(n-1,1),1) + 6*diag(ones(n-1,1),-1);

b = [9; 15*ones(n-2,1); 14];

ut = ones(n,1);
norm_ut = norm(ut);

relerr_Jacobi = zeros(iter+1,1);
relerr_Jacobi(1) = 1;

u = zeros(n,1);
for k = 1:iter
    u_old = u;
    u(1)=0.125*(b(1)-u_old(2));
    for i = 2:n-1
        u(i)=0.125*(b(i)-6*u_old(i-1)-u_old(i+1));
    end
    u(n)=0.125*(b(n)-6*u_old(n-1));
    relerr_Jacobi(k+1) = norm(u - ut) / norm_ut;
end

relerr_GS = zeros(iter+1,1);
relerr_GS(1) = 1;

u = zeros(n,1);
for k = 1:iter
    u(1)=0.125*(b(1)-u(2));
    for i = 2:n-1
        u(i)=0.125*(b(i)-6*u(i-1)-u(i+1));
    end
    u(n)=0.125*(b(n)-6*u(n-1));
    relerr_GS(k+1) = norm(u - ut) / norm_ut;
end

relerr_SOR = zeros(iter+1,1);
relerr_SOR(1) = 1;

omega = 1.11;   
u = zeros(n,1);
for k = 1:iter
    u(1)=(1-omega)*u(1)+omega*0.125*(b(1)-u(2));
    for i = 2:n-1
        u(i)=(1-omega)*u(i)+omega*0.125*(b(i)-6*u(i-1)-u(i+1));
    end
    u(n)=(1-omega)*u(n)+omega*0.125*(b(n)-6*u(n-1));
    relerr_SOR(k+1) = norm(u - ut) / norm_ut;
end


step = 4;
xx = 1:step:iter+1;

relerr_Jacobi = max(relerr_Jacobi, eps);
relerr_GS     = max(relerr_GS,     eps);
relerr_SOR    = max(relerr_SOR,    eps);

figure;
semilogy(xx, relerr_Jacobi(xx), '-b+', ...
         xx, relerr_GS(xx),     '-ro', ...
         xx, relerr_SOR(xx),    '-wd');  
ylim([1e-16 1]);                        
legend('Jacobi','GS','SOR');
title(sprintf('n = %d', n));
xlabel('Iteration');
ylabel('Relative error');
set(gca,'FontSize',12);
