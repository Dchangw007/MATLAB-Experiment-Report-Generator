clear; close all; clc;
rng(0);                                     
n = 100;
A = 100*eye(n) + diag(ones(n-1,1),-1) + diag(ones(n-1,1),1);
A2 = zeros(n,n);
for i = 1 : n
    for j = 1 : n
        A2(i,j) = 1/(i + j + 1 );
    end
end
b = randn(n,1);
b2=zeros(n,1);
for i = 1 : n
    t = 0 ;
    for j = 1 : n
        t = t + 1 / (i + j + 1);
    end
    b ( i )=t;
end

x_ref = A\b;

Lc = Cholesky(A);
y  = forward_sub(Lc, b);                    
x1 = back_sub(Lc', y);                     

[Ld, D] = LDLT(A);
y = forward_sub(Ld, b);                     
z = y ./ D(:);                             
x2 = back_sub(Ld', z);                     

fprintf('--- 残差与相对误差 ---\n');

function L = Cholesky(A)
    n = size(A,1);
    L = zeros(n);
    for j = 1:n
        L(j,j) = A(j,j) - dot(L(j,1:j-1), L(j,1:j-1));  
        L(j,j) = sqrt(L(j,j));
        for i = j+1:n
            L(i,j) = (A(i,j) - dot(L(i,1:j-1), L(j,1:j-1))) / L(j,j);
        end
    end
end

function [L,D] = LDLT(A)
    n = size(A,1);
    L = eye(n);
    D = zeros(1,n);
    for j = 1:n
        D(j) = A(j,j) - dot(L(j,1:j-1).*L(j,1:j-1), D(1:j-1));  
        for i = j+1:n
            L(i,j) = (A(i,j) - dot(L(i,1:j-1).*L(j,1:j-1), D(1:j-1))) / D(j);
        end
    end
end

function y = forward_sub(L, b)
    n = numel(b); y = zeros(n,1);
    for i = 1:n
        y(i) = (b(i) - L(i,1:i-1)*y(1:i-1)) / L(i,i);
    end
end

function x = back_sub(U, b)
    n = numel(b); x = zeros(n,1);
    for i = n:-1:1
        x(i) = (b(i) - U(i,i+1:n)*x(i+1:n)) / U(i,i);
    end
end
